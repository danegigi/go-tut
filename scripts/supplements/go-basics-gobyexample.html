Variables, Constants, and Types

Hello World

The classic first program. Every Go program starts with a package main declaration and a func main() entry point. fmt.Println prints a line to standard output.

package main

import "fmt"

func main() {
    fmt.Println("hello world")
}

Values

Go has various value types including strings, integers, floats, and booleans. Here are a few basic examples.

package main

import "fmt"

func main() {
    fmt.Println("go" + "lang")
    fmt.Println("1+1 =", 1+1)
    fmt.Println("7.0/3.0 =", 7.0/3.0)
    fmt.Println(true && false)
    fmt.Println(true || false)
    fmt.Println(!true)
}

Variables

In Go, variables are explicitly declared. The compiler uses the type to check correctness of function calls. var declares variables; := is shorthand for declare-and-initialize inside functions.

package main

import "fmt"

func main() {
    var a = "initial"
    fmt.Println(a)

    var b, c int = 1, 2
    fmt.Println(b, c)

    var d = true
    fmt.Println(d)

    var e int
    fmt.Println(e) // zero-valued: 0

    f := "apple" // short declaration
    fmt.Println(f)
}

Constants

Go supports constants of character, string, boolean, and numeric values. const declares a constant value that cannot be changed after assignment. Constants can appear anywhere a var can but with the const keyword.

package main

import (
    "fmt"
    "math"
)

const s string = "constant"

func main() {
    fmt.Println(s)
    const n = 500000000
    const d = 3e20 / n
    fmt.Println(d)
    fmt.Println(int64(d))
    fmt.Println(math.Sin(n))
}

Control Flow & Loop Scope Mechanics

For

for is Go's only looping construct. It covers traditional for, while, and infinite loops — there is no while keyword.

package main

import "fmt"

func main() {
    // Classic: init; condition; post
    i := 1
    for i <= 3 {
        fmt.Println(i)
        i = i + 1
    }

    // Traditional for
    for j := 0; j < 3; j++ {
        fmt.Println(j)
    }

    // Infinite loop with break
    for {
        fmt.Println("loop")
        break
    }

    // Continue to next iteration
    for n := 0; n < 6; n++ {
        if n%2 == 0 {
            continue
        }
        fmt.Println(n)
    }
}

If/Else

Branching with if and else. Parentheses are not needed around conditions, but braces are required. You can put a short statement before the condition — its variables are scoped to the if/else block.

package main

import "fmt"

func main() {
    if 7%2 == 0 {
        fmt.Println("7 is even")
    } else {
        fmt.Println("7 is odd")
    }

    if 8%4 == 0 {
        fmt.Println("8 is divisible by 4")
    }

    // Statement before condition
    if num := 9; num < 0 {
        fmt.Println(num, "is negative")
    } else if num < 10 {
        fmt.Println(num, "has 1 digit")
    } else {
        fmt.Println(num, "has multiple digits")
    }
}

Switch

Switch statements express conditionals across many branches. Go's switch is cleaner than C's: it doesn't need break statements and cases don't fall through by default.

package main

import (
    "fmt"
    "time"
)

func main() {
    i := 2
    switch i {
    case 1:
        fmt.Println("one")
    case 2:
        fmt.Println("two")
    case 3:
        fmt.Println("three")
    }

    // Switch on time
    switch time.Now().Weekday() {
    case time.Saturday, time.Sunday:
        fmt.Println("It's the weekend")
    default:
        fmt.Println("It's a weekday")
    }

    // Switch without an expression (like if/else chain)
    t := time.Now()
    switch {
    case t.Hour() < 12:
        fmt.Println("Before noon")
    default:
        fmt.Println("After noon")
    }
}

Functions, Defer, and Error Handling

Functions

Functions are central in Go. They are defined with func and can take zero or more arguments and return zero or more values.

package main

import "fmt"

func plus(a int, b int) int {
    return a + b
}

func plusPlus(a, b, c int) int {
    return a + b + c
}

func main() {
    res := plus(1, 2)
    fmt.Println("1+2 =", res)

    res = plusPlus(1, 2, 3)
    fmt.Println("1+2+3 =", res)
}

Multiple Return Values

Go has built-in support for multiple return values. This is used idiomatically to return both result and error values from a function.

package main

import "fmt"

func vals() (int, int) {
    return 3, 7
}

func main() {
    a, b := vals()
    fmt.Println(a)
    fmt.Println(b)

    // Use _ to ignore a return value
    _, c := vals()
    fmt.Println(c)
}

Variadic Functions

Variadic functions can be called with any number of trailing arguments. fmt.Println is a common variadic function.

package main

import "fmt"

func sum(nums ...int) {
    fmt.Print(nums, " ")
    total := 0
    for _, num := range nums {
        total += num
    }
    fmt.Println(total)
}

func main() {
    sum(1, 2)
    sum(1, 2, 3)

    nums := []int{1, 2, 3, 4}
    sum(nums...) // spread a slice
}

Closures

Go supports anonymous functions, which can form closures. Anonymous functions are useful when you want to define a function inline without naming it. A closure captures variables from its surrounding scope.

package main

import "fmt"

func intSeq() func() int {
    i := 0
    return func() int {
        i++
        return i
    }
}

func main() {
    nextInt := intSeq()
    fmt.Println(nextInt()) // 1
    fmt.Println(nextInt()) // 2
    fmt.Println(nextInt()) // 3

    newInts := intSeq()
    fmt.Println(newInts()) // 1 — fresh closure
}

Recursion

Go supports recursive functions. A classic example is factorial.

package main

import "fmt"

func fact(n int) int {
    if n == 0 {
        return 1
    }
    return n * fact(n-1)
}

func main() {
    fmt.Println(fact(7)) // 5040

    // Recursive anonymous function (closure)
    var fib func(n int) int
    fib = func(n int) int {
        if n < 2 {
            return n
        }
        return fib(n-1) + fib(n-2)
    }
    fmt.Println(fib(7)) // 13
}

Defer

defer ensures a function call is performed later, usually for cleanup. It's commonly used to close files, unlock mutexes, or print a footer. Deferred calls execute in LIFO order when the surrounding function returns.

package main

import (
    "fmt"
    "os"
)

func main() {
    f := createFile("/tmp/defer.txt")
    defer closeFile(f)
    writeFile(f)
}

func createFile(p string) *os.File {
    fmt.Println("creating")
    f, err := os.Create(p)
    if err != nil {
        panic(err)
    }
    return f
}

func writeFile(f *os.File) {
    fmt.Println("writing")
    fmt.Fprintln(f, "data")
}

func closeFile(f *os.File) {
    fmt.Println("closing")
    err := f.Close()
    if err != nil {
        fmt.Fprintf(os.Stderr, "error: %v\n", err)
        os.Exit(1)
    }
}

Panic

A panic means something went unexpectedly wrong. Typically we use it for errors that shouldn't occur during normal operation — programmer errors, not user input errors.

package main

import "os"

func main() {
    panic("a problem")

    // A common use: abort if a function returns an unexpected error
    _, err := os.Create("/tmp/file")
    if err != nil {
        panic(err)
    }
}

Recover

recover catches a panic and resumes normal execution. It must be called inside a deferred function. This is useful for libraries that shouldn't crash the caller on unexpected errors.

package main

import "fmt"

func mayPanic() {
    panic("a problem")
}

func safeDiv(a, b int) (result int, err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("recovered: %v", r)
        }
    }()
    return a / b, nil
}

func main() {
    // Recover from a panic
    func() {
        defer func() {
            if r := recover(); r != nil {
                fmt.Println("Recovered:", r)
            }
        }()
        mayPanic()
    }()

    fmt.Println("After mayPanic()")

    // Practical use: safe division
    result, err := safeDiv(10, 0)
    fmt.Println("Result:", result, "Error:", err)
}

Errors

In Go, errors are values. Functions communicate errors by returning a value of type error as their last return value. A nil error means success; a non-nil error carries a message describing what went wrong.

package main

import (
    "errors"
    "fmt"
)

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("cannot divide by zero")
    }
    return a / b, nil
}

func main() {
    result, err := divide(10, 3)
    if err != nil {
        fmt.Println("error:", err)
    } else {
        fmt.Printf("10 / 3 = %.2f\n", result)
    }

    // This will produce an error
    result, err = divide(10, 0)
    if err != nil {
        fmt.Println("error:", err)
    } else {
        fmt.Printf("10 / 0 = %.2f\n", result)
    }
}

Custom Errors

You can create custom error types by implementing the error interface (which requires only an Error() string method). Custom errors let callers inspect error details using type assertions or errors.As.

package main

import (
    "errors"
    "fmt"
)

// ArgError is a custom error type with extra context
type ArgError struct {
    Arg     int
    Message string
}

func (e *ArgError) Error() string {
    return fmt.Sprintf("%d - %s", e.Arg, e.Message)
}

func doSomething(arg int) (int, error) {
    if arg == 42 {
        return -1, &ArgError{Arg: arg, Message: "can't work with it"}
    }
    return arg + 3, nil
}

func main() {
    _, err := doSomething(42)
    if err != nil {
        // Use errors.As to check for a specific error type
        var ae *ArgError
        if errors.As(err, &ae) {
            fmt.Println("arg:", ae.Arg)
            fmt.Println("message:", ae.Message)
        } else {
            fmt.Println("unexpected error:", err)
        }
    }
}

Data Structures and Mutation

Arrays

An array is a numbered sequence of elements of a fixed length. Arrays are values in Go — assigning one copies all elements. In typical Go code, slices are much more common than arrays.

package main

import "fmt"

func main() {
    var a [5]int
    fmt.Println("emp:", a)

    a[4] = 100
    fmt.Println("set:", a)
    fmt.Println("get:", a[4])
    fmt.Println("len:", len(a))

    b := [5]int{1, 2, 3, 4, 5}
    fmt.Println("dcl:", b)

    // 2D array
    var twoD [2][3]int
    for i := 0; i < 2; i++ {
        for j := 0; j < 3; j++ {
            twoD[i][j] = i + j
        }
    }
    fmt.Println("2d:", twoD)
}

Slices

Slices are a key data type in Go, giving a more powerful interface than arrays. Unlike arrays, slices are dynamically-sized. They have a length and a capacity, backed by an underlying array.

package main

import "fmt"

func main() {
    s := make([]string, 3)
    s[0] = "a"
    s[1] = "b"
    s[2] = "c"
    fmt.Println("set:", s)
    fmt.Println("get:", s[2])
    fmt.Println("len:", len(s))

    s = append(s, "d")
    s = append(s, "e", "f")
    fmt.Println("apd:", s)

    c := make([]string, len(s))
    copy(c, s)
    fmt.Println("cpy:", c)

    l := s[2:5]
    fmt.Println("sl1:", l)

    t := []string{"g", "h", "i"}
    fmt.Println("dcl:", t)
}

Maps

Maps are Go's built-in associative data type (sometimes called hashes or dicts in other languages). They map keys to values and are created with make or a literal.

package main

import "fmt"

func main() {
    m := make(map[string]int)
    m["k1"] = 7
    m["k2"] = 13
    fmt.Println("map:", m)
    fmt.Println("len:", len(m))

    v1 := m["k1"]
    fmt.Println("v1:", v1)

    delete(m, "k2")
    fmt.Println("map:", m)

    // The optional second return value indicates presence
    _, prs := m["k2"]
    fmt.Println("prs:", prs)

    n := map[string]int{"foo": 1, "bar": 2}
    fmt.Println("map:", n)
}

Sorting

Go's slices package implements sorting for built-in and user-defined types.

package main

import (
    "fmt"
    "slices"
)

func main() {
    strs := []string{"c", "a", "b"}
    slices.Sort(strs)
    fmt.Println("Strings:", strs)

    ints := []int{7, 2, 4}
    slices.Sort(ints)
    fmt.Println("Ints:   ", ints)

    fmt.Println("Sorted: ", slices.IsSorted(ints))
}

Sorting by Functions

Sometimes you want to sort a collection by something other than its natural order. Use slices.SortFunc with a custom comparison function.

package main

import (
    "cmp"
    "fmt"
    "slices"
)

func main() {
    fruits := []string{"peach", "kiwi", "apple", "pear"}
    // Sort by length
    lenCmp := func(a, b string) int {
        return cmp.Compare(len(a), len(b))
    }
    slices.SortFunc(fruits, lenCmp)
    fmt.Println(fruits) // [kiwi pear peach apple]

    type Person struct {
        Name string
        Age  int
    }
    people := []Person{
        {"Jax", 37}, {"TJ", 25}, {"Alex", 72},
    }
    slices.SortFunc(people, func(a, b Person) int {
        return cmp.Compare(a.Age, b.Age)
    })
    fmt.Println(people)
}

Range over Built-in Types

range iterates over elements in a variety of built-in data structures. Let's see how to use range with slices, maps, and strings.

package main

import "fmt"

func main() {
    // Range over a slice
    nums := []int{2, 3, 4}
    sum := 0
    for _, num := range nums {
        sum += num
    }
    fmt.Println("sum:", sum)

    // Range with index
    for i, num := range nums {
        if num == 3 {
            fmt.Println("index:", i)
        }
    }

    // Range over a map
    kvs := map[string]string{"a": "apple", "b": "banana"}
    for k, v := range kvs {
        fmt.Printf("%s -> %s\n", k, v)
    }

    // Range over a string yields Unicode code points
    for i, c := range "go" {
        fmt.Println(i, c)
    }
}

Structs, Methods, and Interfaces

Pointers

Go supports pointers, allowing you to pass references to values and records within your program. A pointer holds the memory address of a value. The & operator generates a pointer to its operand, and the * operator dereferences a pointer.

package main

import "fmt"

// zeroval gets a copy of ival -- changes don't affect the caller
func zeroval(ival int) {
    ival = 0
}

// zeroptr has a *int parameter, meaning it takes an int pointer.
// It dereferences the pointer to set the value at that address.
func zeroptr(iptr *int) {
    *iptr = 0
}

func main() {
    i := 1
    fmt.Println("initial:", i)

    zeroval(i)
    fmt.Println("zeroval:", i)

    // &i gives the memory address of i
    zeroptr(&i)
    fmt.Println("zeroptr:", i)

    // Pointers can be printed too
    fmt.Println("pointer:", &i)
}

Strings and Runes

A Go string is a read-only slice of bytes. The language and standard library treat strings specially — as containers of text encoded in UTF-8. A rune is an alias for int32 and represents a Unicode code point.

package main

import (
    "fmt"
    "unicode/utf8"
)

func main() {
    // s is a string assigned a literal value representing
    // the word "hello" in Thai + English
    s := "สวัสดี hello"

    // Length in bytes (not characters)
    fmt.Println("Len:", len(s))

    // Length in runes (characters)
    fmt.Println("Rune count:", utf8.RuneCountInString(s))

    // Iterating over a string with range yields runes
    for i, r := range s {
        fmt.Printf("%d\t%c\t%U\n", i, r, r)
    }

    // You can also decode runes manually
    fmt.Println("\nManual decoding:")
    for i := 0; i < len(s); {
        r, size := utf8.DecodeRuneInString(s[i:])
        fmt.Printf("%d\t%c\n", i, r)
        i += size
    }
}

Structs

Go's structs are typed collections of fields. They're useful for grouping data together to form records. Structs are value types — when assigned or passed, they are copied.

package main

import "fmt"

type person struct {
    name string
    age  int
}

// newPerson constructs a new person struct with the given name.
func newPerson(name string) *person {
    p := person{name: name}
    p.age = 42
    return &p
}

func main() {
    // Create a new struct
    fmt.Println(person{"Bob", 20})

    // Name fields at initialization
    fmt.Println(person{name: "Alice", age: 30})

    // Omitted fields are zero-valued
    fmt.Println(person{name: "Fred"})

    // An & prefix yields a pointer to the struct
    fmt.Println(&person{name: "Ann", age: 40})

    // Use a constructor function
    fmt.Println(newPerson("Jon"))

    // Access struct fields with a dot
    s := person{name: "Sean", age: 50}
    fmt.Println(s.name)

    // Pointers to structs are automatically dereferenced
    sp := &s
    fmt.Println(sp.age)

    // Structs are mutable
    sp.age = 51
    fmt.Println(sp.age)
}

Methods

Go supports methods defined on struct types. A method is a function with a special receiver argument. You can define methods with pointer receivers (to mutate state) or value receivers (read-only).

package main

import "fmt"

type rect struct {
    width, height int
}

// This area method has a value receiver of type rect
func (r rect) area() int {
    return r.width * r.height
}

// Methods can have pointer receivers to mutate the struct
func (r rect) perim() int {
    return 2*r.width + 2*r.height
}

// scale uses a pointer receiver so it can modify the struct
func (r *rect) scale(factor int) {
    r.width *= factor
    r.height *= factor
}

func main() {
    r := rect{width: 10, height: 5}

    fmt.Println("area:", r.area())
    fmt.Println("perim:", r.perim())

    // Go automatically handles conversion between values and
    // pointers for method calls
    rp := &r
    fmt.Println("area:", rp.area())

    r.scale(2)
    fmt.Println("after scale, area:", r.area())
}

Interfaces

Interfaces are named collections of method signatures. They are implemented implicitly — a type implements an interface by implementing its methods. No explicit "implements" declaration is needed.

package main

import (
    "fmt"
    "math"
)

// geometry is an interface type
type geometry interface {
    area() float64
    perim() float64
}

type circle struct {
    radius float64
}

type rectangle struct {
    width, height float64
}

// rectangle implements geometry
func (r rectangle) area() float64 {
    return r.width * r.height
}
func (r rectangle) perim() float64 {
    return 2*r.width + 2*r.height
}

// circle implements geometry
func (c circle) area() float64 {
    return math.Pi * c.radius * c.radius
}
func (c circle) perim() float64 {
    return 2 * math.Pi * c.radius
}

// measure works with any geometry
func measure(g geometry) {
    fmt.Println(g)
    fmt.Println("area:", g.area())
    fmt.Println("perim:", g.perim())
}

func main() {
    r := rectangle{width: 3, height: 4}
    c := circle{radius: 5}

    measure(r)
    measure(c)
}

Enums

Go doesn't have a dedicated enum keyword but uses typed constants with iota to create enumerated types. This pattern gives you type safety and auto-incrementing values.

package main

import "fmt"

// ServerState is a custom type for our enum
type ServerState int

const (
    StateIdle ServerState = iota
    StateConnected
    StateError
    StateRetrying
)

// Implement the Stringer interface for readable output
func (ss ServerState) String() string {
    switch ss {
    case StateIdle:
        return "idle"
    case StateConnected:
        return "connected"
    case StateError:
        return "error"
    case StateRetrying:
        return "retrying"
    default:
        return fmt.Sprintf("unknown(%d)", ss)
    }
}

func transition(s ServerState) ServerState {
    switch s {
    case StateIdle:
        return StateConnected
    case StateConnected, StateRetrying:
        return StateIdle
    case StateError:
        return StateRetrying
    default:
        panic(fmt.Sprintf("unknown state: %s", s))
    }
}

func main() {
    ns := transition(StateIdle)
    fmt.Println(ns) // connected

    ns2 := transition(ns)
    fmt.Println(ns2) // idle
}

Struct Embedding

Go supports embedding of structs and interfaces to express a more seamless composition of types. This is not inheritance — it's composition where the embedded type's methods are promoted to the embedding struct.

package main

import "fmt"

type base struct {
    num int
}

func (b base) describe() string {
    return fmt.Sprintf("base with num=%d", b.num)
}

// container embeds base — it gets base's fields and methods
type container struct {
    base
    str string
}

func main() {
    co := container{
        base: base{num: 1},
        str:  "some name",
    }

    // We can access base's fields directly on container
    fmt.Printf("co={num: %d, str: %s}\n", co.num, co.str)

    // The describe method from base is promoted
    fmt.Println("describe:", co.describe())

    // The embedding is explicit when using struct literal
    type describer interface {
        describe() string
    }

    // container implements describer via its embedded base
    var d describer = co
    fmt.Println("describer:", d.describe())
}

Generics

Generics

Starting with Go 1.18, the language supports generics via type parameters. This lets you write functions and types that work with multiple types while remaining type-safe at compile time.

package main

import "fmt"

// MapKeys returns the keys of any map. The type parameters
// are specified in square brackets.
func MapKeys[K comparable, V any](m map[K]V) []K {
    r := make([]K, 0, len(m))
    for k := range m {
        r = append(r, k)
    }
    return r
}

// List is a generic singly-linked list
type List[T any] struct {
    head, tail *element[T]
}

type element[T any] struct {
    next *element[T]
    val  T
}

func (lst *List[T]) Push(v T) {
    if lst.tail == nil {
        lst.head = &element[T]{val: v}
        lst.tail = lst.head
    } else {
        lst.tail.next = &element[T]{val: v}
        lst.tail = lst.tail.next
    }
}

func (lst *List[T]) GetAll() []T {
    var elems []T
    for e := lst.head; e != nil; e = e.next {
        elems = append(elems, e.val)
    }
    return elems
}

func main() {
    var m = map[int]string{1: "2", 2: "4", 4: "8"}
    fmt.Println("keys:", MapKeys(m))

    lst := List[int]{}
    lst.Push(10)
    lst.Push(13)
    lst.Push(23)
    fmt.Println("list:", lst.GetAll())
}

Range over Iterators

Starting with Go 1.23, you can use range over custom iterator functions. An iterator is a function that takes a yield callback and calls it for each element. This enables for...range syntax over any sequence.

package main

import (
    "fmt"
    "iter"
)

// Fibonacci returns an iterator that yields Fibonacci numbers
func Fibonacci() iter.Seq[int] {
    return func(yield func(int) bool) {
        a, b := 0, 1
        for {
            if !yield(a) {
                return
            }
            a, b = b, a+b
        }
    }
}

func main() {
    // Range over a custom iterator
    for n := range Fibonacci() {
        if n > 100 {
            break
        }
        fmt.Println(n)
    }
}

Concurrency Patterns

Channel Buffering

By default channels are unbuffered — they only accept sends if there's a corresponding receive ready. Buffered channels accept a limited number of values without a corresponding receiver.

package main

import "fmt"

func main() {
    messages := make(chan string, 2)
    messages <- "buffered"
    messages <- "channel"
    fmt.Println(<-messages)
    fmt.Println(<-messages)
}

Channel Synchronization

We can use channels to synchronize execution across goroutines. A blocking receive waits for the sender to complete.

package main

import (
    "fmt"
    "time"
)

func worker(done chan bool) {
    fmt.Print("working...")
    time.Sleep(time.Second)
    fmt.Println("done")
    done <- true
}

func main() {
    done := make(chan bool, 1)
    go worker(done)
    <-done
}

Channel Directions

When using channels as function parameters, you can specify if a channel is meant to only send or receive values, increasing type-safety.

package main

import "fmt"

func ping(pings chan<- string, msg string) {
    pings <- msg
}

func pong(pings <-chan string, pongs chan<- string) {
    msg := <-pings
    pongs <- msg
}

func main() {
    pings := make(chan string, 1)
    pongs := make(chan string, 1)
    ping(pings, "passed message")
    pong(pings, pongs)
    fmt.Println(<-pongs)
}

Select

Go's select lets you wait on multiple channel operations. It's like a switch but for channels — it blocks until one case can proceed.

package main

import (
    "fmt"
    "time"
)

func main() {
    c1 := make(chan string)
    c2 := make(chan string)

    go func() {
        time.Sleep(1 * time.Second)
        c1 <- "one"
    }()
    go func() {
        time.Sleep(2 * time.Second)
        c2 <- "two"
    }()

    for i := 0; i < 2; i++ {
        select {
        case msg1 := <-c1:
            fmt.Println("Received", msg1)
        case msg2 := <-c2:
            fmt.Println("Received", msg2)
        }
    }
}

Non-Blocking Channel Operations

A select with a default case implements a non-blocking send, receive, or multi-way select.

package main

import "fmt"

func main() {

    case msg := <-messages:

    case messages <- msg:

        fmt.Println("received signal", sig)

Closing Channels

Closing a channel indicates that no more values will be sent on it. This is useful to communicate completion to a receiver.

package main

}

Worker Pools

A worker pool is a common concurrency pattern: a fixed number of goroutines process jobs from a shared channel. This limits concurrency while keeping throughput high.

package main

import (
    "fmt"
    "time"
)

func worker(id int, jobs <-chan int, results chan<- int) {
    for j := range jobs {
        fmt.Println("worker", id, "started  job", j)
        time.Sleep(time.Second)
        fmt.Println("worker", id, "finished job", j)
        results <- j * 2
    }
}

func main() {
    const numJobs = 5
    jobs := make(chan int, numJobs)
    results := make(chan int, numJobs)

    for w := 1; w <= 3; w++ {
        go worker(w, jobs, results)
    }

    for j := 1; j <= numJobs; j++ {
        jobs <- j
    }
    close(jobs)

    for a := 1; a <= numJobs; a++ {
        <-results
    }
}

WaitGroups

A sync.WaitGroup waits for a collection of goroutines to finish. Add to the counter before launching, call Done when each finishes, and Wait blocks until the counter is zero.

package main

import (
    "fmt"
    "sync"
    "time"
)

func workerWG(id int) {
    fmt.Printf("Worker %d starting\n", id)
    time.Sleep(time.Second)
    fmt.Printf("Worker %d done\n", id)
}

func main() {
    var wg sync.WaitGroup

    for i := 1; i <= 5; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            workerWG(id)
        }(i)
    }
    wg.Wait()
}

Rate Limiting

Rate limiting is an important mechanism for controlling resource utilization and maintaining quality of service. Go supports rate limiting with goroutines, channels, and tickers.

package main

import (
    "fmt"
    "time"
)

func main() {
    requests := make(chan int, 5)
    for i := 1; i <= 5; i++ {
        requests <- i
    }
    close(requests)

    limiter := time.Tick(200 * time.Millisecond)

    for req := range requests {
        <-limiter
        fmt.Println("request", req, time.Now())
    }

    // Bursty limiter: allow short bursts
    burstyLimiter := make(chan time.Time, 3)
    for i := 0; i < 3; i++ {
        burstyLimiter <- time.Now()
    }
    go func() {
        for t := range time.Tick(200 * time.Millisecond) {
            burstyLimiter <- t
        }
    }()

    burstyRequests := make(chan int, 5)
    for i := 1; i <= 5; i++ {
        burstyRequests <- i
    }
    close(burstyRequests)
    for req := range burstyRequests {
        <-burstyLimiter
        fmt.Println("request", req, time.Now())
    }
}

Atomic Counters

The sync/atomic package provides low-level atomic operations on integers, useful for simple shared state accessed by multiple goroutines without a full mutex.

package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

    var wg sync.WaitGroup

Mutexes

For more complex shared state, use a sync.Mutex to safely access data across multiple goroutines.

package main

import (
    "fmt"
    "sync"
)

type Container struct {
    mu       sync.Mutex
    counters map[string]int
}

func (c *Container) inc(name string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.counters[name]++
}

func main() {
    c := Container{counters: map[string]int{"a": 0, "b": 0}}
    var wg sync.WaitGroup

    doIncrement := func(name string, n int) {
        for i := 0; i < n; i++ {
            c.inc(name)
        }
        wg.Done()
    }

    wg.Add(3)
    go doIncrement("a", 10000)
    go doIncrement("a", 10000)
    go doIncrement("b", 10000)
    wg.Wait()
    fmt.Println(c.counters) // map[a:20000 b:10000]
}

Stateful Goroutines

An alternative to mutexes: use a single goroutine that owns the state, and communicate via channels. This is the "share memory by communicating" approach.

package main

import (
    "fmt"
    "math/rand"
    "sync/atomic"
    "time"
)

type readOp struct {
    key  int
    resp chan int
}
type writeOp struct {
    key  int
    val  int
    resp chan bool
}

func main() {
    var readOps, writeOps atomic.Uint64
    reads := make(chan readOp)
    writes := make(chan writeOp)

    // State-owning goroutine
    go func() {
        state := make(map[int]int)
        for {
            select {
            case read := <-reads:
                read.resp <- state[read.key]
            case write := <-writes:
                state[write.key] = write.val
                write.resp <- true
            }
        }
    }()

    // 100 reads
    for r := 0; r < 100; r++ {
        go func() {
            for {
                read := readOp{key: rand.Intn(5), resp: make(chan int)}
                reads <- read
                <-read.resp
                readOps.Add(1)
                time.Sleep(time.Millisecond)
            }
        }()
    }
    // 10 writes
    for w := 0; w < 10; w++ {
        go func() {
            for {
                write := writeOp{key: rand.Intn(5), val: rand.Intn(100), resp: make(chan bool)}
                writes <- write
                <-write.resp
                writeOps.Add(1)
                time.Sleep(time.Millisecond)
            }
        }()
    }

    time.Sleep(time.Second)
    fmt.Println("readOps:", readOps.Load())
    fmt.Println("writeOps:", writeOps.Load())
}

Context

context.Context carries deadlines, cancellation signals, and request-scoped values across API boundaries and goroutines. It's the standard way to control long-running operations.

Goroutines

A goroutine is a lightweight thread of execution managed by the Go runtime. You start one by placing go before a function call. Goroutines run concurrently with the calling code and with each other.

package main

import (
    "fmt"
    "time"
)

func worker(id int) {
    fmt.Printf("Worker %d starting\n", id)
    time.Sleep(time.Second)
    fmt.Printf("Worker %d done\n", id)
}

func main() {
    // Start several goroutines
    for i := 1; i <= 5; i++ {
        go worker(i)
    }

    // You can also start a goroutine with an anonymous function
    go func(msg string) {
        fmt.Println(msg)
    }("going")

    // Wait for goroutines to finish (in real code, use sync.WaitGroup)
    time.Sleep(2 * time.Second)
    fmt.Println("all workers done")
}

Channels

Channels are the pipes that connect concurrent goroutines. You can send values into channels from one goroutine and receive those values in another. By default, sends and receives block until both sides are ready, enabling synchronization without locks.

package main

import "fmt"

func main() {
    // Create a new channel with make(chan val-type)
    messages := make(chan string)

    // Send a value into a channel from a goroutine
    go func() {
        messages <- "ping"
    }()

    // Receive a value from the channel
    msg := <-messages
    fmt.Println(msg)

    // Buffered channels accept a limited number of values
    // without a corresponding receiver
    ch := make(chan int, 2)
    ch <- 1
    ch <- 2

    fmt.Println(<-ch)
    fmt.Println(<-ch)

    // Channels can be used for synchronization
    done := make(chan bool)
    go func() {
        fmt.Println("working...")
        done <- true
    }()
    <-done
    fmt.Println("finished")
}

URL Parsing

URLs provide a uniform way to locate resources. The net/url package parses URLs into their components.

package main

import (
    "fmt"
    "net"
    "net/url"
)

func main() {
    s := "postgres://user:pass@host.com:5432/path?k=v#f"
    u, err := url.Parse(s)
    if err != nil {
        panic(err)
    }
    fmt.Println(u.Scheme)
    fmt.Println(u.User)
    fmt.Println(u.User.Username())
    p, _ := u.User.Password()
    fmt.Println(p)
    fmt.Println(u.Host)
    host, port, _ := net.SplitHostPort(u.Host)
    fmt.Println(host, port)
    fmt.Println(u.Path)
    fmt.Println(u.Fragment)
    fmt.Println(u.RawQuery)
    m, _ := url.ParseQuery(u.RawQuery)
    fmt.Println(m)
    fmt.Println(m["k"][0])
}

}

Reading Files

Reading files is one of the most basic tasks. Go's os package provides the tools.

package main

import (
    "fmt"
    "os"
)

func check(e error) {
    if e != nil {
        panic(e)
    }
}

func main() {
    dat, err := os.ReadFile("/tmp/dat")
    check(err)
    fmt.Print(string(dat))

    f, err := os.Open("/tmp/dat")
    check(err)
    defer f.Close()

    b1 := make([]byte, 5)
    n1, err := f.Read(b1)
    check(err)
    fmt.Printf("%d bytes: %s\n", n1, string(b1[:n1]))

    o2, err := f.Seek(6, 0)
    check(err)
    b2 := make([]byte, 2)
    n2, err := f.Read(b2)
    check(err)
    fmt.Printf("%d bytes @ %d: %s\n", n2, o2, string(b2[:n2]))
}

Writing Files

Writing files in Go follows similar patterns to reading. Go provides multiple approaches from simple one-shot writes to buffered writing for performance.

package main

import (
    "bufio"
    "fmt"
    "os"
)

func main() {
    d1 := []byte("hello\ngo\n")
    err := os.WriteFile("/tmp/dat1", d1, 0644)
    if err != nil { panic(err) }

    f, err := os.Create("/tmp/dat2")
    if err != nil { panic(err) }
    defer f.Close()

    d2 := []byte{115, 111, 109, 101, 10}
    n2, err := f.Write(d2)
    fmt.Printf("wrote %d bytes\n", n2)

    n3, err := f.WriteString("writes\n")
    fmt.Printf("wrote %d bytes\n", n3)

    f.Sync()

    w := bufio.NewWriter(f)
    n4, err := w.WriteString("buffered\n")
    fmt.Printf("wrote %d bytes\n", n4)
    w.Flush()
}

Line Filters

A line filter reads input on stdin, processes it, and prints results to stdout. Uses bufio.Scanner.

package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

func main() {
    scanner := bufio.NewScanner(os.Stdin)
    for scanner.Scan() {
        ucl := strings.ToUpper(scanner.Text())
        fmt.Println(ucl)
    }
    if err := scanner.Err(); err != nil {
        fmt.Fprintln(os.Stderr, "error:", err)
        os.Exit(1)
    }
}

File Paths

The filepath package provides portable path manipulation — Join, Dir, Base, Ext, Rel, IsAbs.

package main

import (
    "fmt"
    "path/filepath"
    "strings"
)

func main() {
    p := filepath.Join("dir1", "dir2", "filename")
    fmt.Println("p:", p)

    fmt.Println(filepath.Join("dir1//", "filename"))
    fmt.Println(filepath.Join("dir1/../dir1", "filename"))

    fmt.Println("Dir(p):", filepath.Dir(p))
    fmt.Println("Base(p):", filepath.Base(p))

    fmt.Println(filepath.IsAbs("dir/file"))
    fmt.Println(filepath.IsAbs("/dir/file"))

    filename := "config.json"
    ext := filepath.Ext(filename)
    fmt.Println(ext)
    fmt.Println(strings.TrimSuffix(filename, ext))

    rel, _ := filepath.Rel("a/b", "a/b/t/file")
    fmt.Println(rel)

    rel, _ = filepath.Rel("a/b", "a/c/t/file")
    fmt.Println(rel)
}

Directories

Create, list, and walk directories using os.Mkdir, os.MkdirAll, os.ReadDir, and filepath.WalkDir.

package main

import (
    "fmt"
    "io/fs"
    "os"
    "path/filepath"
)

func check(e error) { if e != nil { panic(e) } }

func main() {
    err := os.Mkdir("subdir", 0755)
    check(err)
    defer os.RemoveAll("subdir")

    createEmptyFile := func(name string) {
        check(os.WriteFile(name, []byte(""), 0644))
    }
    createEmptyFile("subdir/file1")

    os.MkdirAll("subdir/parent/child", 0755)
    createEmptyFile("subdir/parent/file2")
    createEmptyFile("subdir/parent/child/file4")

    c, _ := os.ReadDir("subdir/parent")
    fmt.Println("Listing subdir/parent")
    for _, entry := range c {
        fmt.Println(" ", entry.Name(), entry.IsDir())
    }

    filepath.WalkDir("subdir", func(path string, d fs.DirEntry, err error) error {
        if err != nil { return err }
        fmt.Println(" ", path, d.IsDir())
        return nil
    })
}

Temporary Files and Directories

os.CreateTemp and os.MkdirTemp create temporary files/dirs. The "*" in patterns is replaced with a random value.

package main

import (
    "fmt"
    "os"
    "path/filepath"
)

func main() {
    f, err := os.CreateTemp("", "sample")
    if err != nil { panic(err) }
    fmt.Println("Temp file:", f.Name())
    defer os.Remove(f.Name())

    f.Write([]byte{1, 2, 3, 4})

    dname, err := os.MkdirTemp("", "sampledir")
    if err != nil { panic(err) }
    fmt.Println("Temp dir:", dname)
    defer os.RemoveAll(dname)

    fname := filepath.Join(dname, "file1")
    os.WriteFile(fname, []byte{1, 2}, 0666)
}

Environment Variables

os.Setenv, os.Getenv, os.LookupEnv (distinguishes unset from empty), and os.Environ (all vars).

package main

import (
    "fmt"
    "os"
    "strings"
)

func main() {
    os.Setenv("FOO", "1")
    fmt.Println("FOO:", os.Getenv("FOO"))
    fmt.Println("BAR:", os.Getenv("BAR"))

    val, ok := os.LookupEnv("BAR")
    if !ok {
        fmt.Println("BAR is not set")
    } else {
        fmt.Println("BAR:", val)
    }

    fmt.Println()
    for _, e := range os.Environ() {
        pair := strings.SplitN(e, "=", 2)
        fmt.Println(pair[0])
    }
}

Spawning Processes

Sometimes a Go program needs to spawn other processes. os/exec runs external commands and captures their output.

package main

import (
    "fmt"
    "os/exec"
)

func main() {
    // Simple command
    out, err := exec.Command("date").Output()
    if err != nil {
        panic(err)
    }
    fmt.Println("date:", string(out))

    // Command with args + piping
    grep := exec.Command("grep", "hello")
    grep.Stdin = nil
    grepIn, _ := grep.StdinPipe()
    grepOut, _ := grep.StdoutPipe()
    grep.Start()
    grepIn.Write([]byte("hello grep\ngoodbye grep"))
    grepIn.Close()
    grepBytes, _ := fmt.Println(grepOut)
    _ = grepBytes
    grep.Wait()
}

Exec'ing Processes

Sometimes we want to completely replace the current Go process with another. syscall.Exec does this (Unix only) — the Go program disappears and the new process takes over its PID.

package main

import (
    "os"
    "os/exec"
    "syscall"
)

func main() {
    binary, err := exec.LookPath("ls")
    if err != nil {
        panic(err)
    }
    args := []string{"ls", "-a", "-l", "-h"}
    env := os.Environ()
    execErr := syscall.Exec(binary, args, env)
    if execErr != nil {
        panic(execErr)
    }
}

Signals

Go programs can handle Unix signals for graceful shutdown. The os/signal package notifies a channel when signals arrive.

package main

import (
    "fmt"
    "os"
    "os/signal"
    "syscall"
)

func main() {
    sigs := make(chan os.Signal, 1)
    signal.Notify(sigs, syscall.SIGINT, syscall.SIGTERM)

    done := make(chan bool, 1)
    go func() {
        sig := <-sigs
        fmt.Println()
        fmt.Println("received:", sig)
        done <- true
    }()

    fmt.Println("awaiting signal (press Ctrl+C)")
    <-done
    fmt.Println("exiting")
}

Exit

Use os.Exit to immediately exit with a status code. Deferred functions will NOT run. Use this for scripts/CLI tools, not libraries.

package main

import (
    "fmt"
    "os"
)

func main() {
    defer fmt.Println("!") // this will NOT be printed

}

HTTP Server

Writing a basic HTTP server in Go is straightforward. Handlers are functions with a standard signature. The server runs until stopped.

package main

import (
    "fmt"
    "net/http"
)

func hello(w http.ResponseWriter, req *http.Request) {
    fmt.Fprintf(w, "hello\n")
}

func headers(w http.ResponseWriter, req *http.Request) {
    for name, headers := range req.Header {
        for _, h := range headers {
            fmt.Fprintf(w, "%v: %v\n", name, h)
        }
    }
}

func main() {
    http.HandleFunc("/hello", hello)
    http.HandleFunc("/headers", headers)
    fmt.Println("Listening on :8090")
    http.ListenAndServe(":8090", nil)
}

TCP Server

Go's net package provides low-level networking. Here's a TCP echo server that handles each connection in its own goroutine.

package main

import (
    "fmt"
    "io"
    "net"
)

func handleConn(conn net.Conn) {
    defer conn.Close()
    io.Copy(conn, conn) // echo back
}

func main() {
    ln, err := net.Listen("tcp", ":9000")
    if err != nil {
        panic(err)
    }
    fmt.Println("TCP echo server on :9000")
    for {
        conn, err := ln.Accept()
        if err != nil {
            continue
        }
        go handleConn(conn)
    }
}

Command-Line Flags

The flag package handles typed flags with defaults, descriptions, and auto-generated -h output.

package main

import (
    "flag"
    "fmt"
)

func main() {
    wordPtr := flag.String("word", "foo", "a string")
    numbPtr := flag.Int("numb", 42, "an int")
    forkPtr := flag.Bool("fork", false, "a bool")

    var svar string
    flag.StringVar(&svar, "svar", "bar", "a string var")

    flag.Parse()

    fmt.Println("word:", *wordPtr)
    fmt.Println("numb:", *numbPtr)
    fmt.Println("fork:", *forkPtr)
    fmt.Println("svar:", svar)
    fmt.Println("tail:", flag.Args())
}

Command-Line Subcommands

Use flag.NewFlagSet for each subcommand, then parse based on os.Args[1].

package main

import (
    "flag"
    "fmt"
    "os"
)

func main() {
    fooCmd := flag.NewFlagSet("foo", flag.ExitOnError)
    fooEnable := fooCmd.Bool("enable", false, "enable")
    fooName := fooCmd.String("name", "", "name")

    barCmd := flag.NewFlagSet("bar", flag.ExitOnError)
    barLevel := barCmd.Int("level", 0, "level")

    if len(os.Args) < 2 {
        fmt.Println("expected 'foo' or 'bar' subcommands")
        os.Exit(1)
    }

    switch os.Args[1] {
    case "foo":
        fooCmd.Parse(os.Args[2:])
        fmt.Println("subcommand 'foo'")
        fmt.Println("  enable:", *fooEnable)
        fmt.Println("  name:", *fooName)
        fmt.Println("  tail:", fooCmd.Args())
    case "bar":
        barCmd.Parse(os.Args[2:])
        fmt.Println("subcommand 'bar'")
        fmt.Println("  level:", *barLevel)
        fmt.Println("  tail:", barCmd.Args())
    default:
        fmt.Println("expected 'foo' or 'bar' subcommands")
        os.Exit(1)
    }
}
